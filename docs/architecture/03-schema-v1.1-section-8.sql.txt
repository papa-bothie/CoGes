  CREATE OR REPLACE FUNCTION set_updated_at() RETURNS trigger AS $$
  BEGIN NEW.updated_at := now(); RETURN NEW; END $$ LANGUAGE plpgsql;

  -- Formule CMP unique, partagee par le trigger et la reconstruction :
  -- aucune divergence possible entre temps reel et recalcul.
  CREATE OR REPLACE FUNCTION cmp_next_avg(
     p_qty numeric, p_avg numeric, p_delta numeric, p_cost numeric)
  RETURNS numeric LANGUAGE sql IMMUTABLE AS $$
     SELECT CASE
        WHEN p_delta <= 0              THEN p_avg                      -- sortie : inchange
        WHEN p_qty + p_delta = 0       THEN p_avg                      -- garde-fou 0/0
        ELSE round((p_qty * p_avg + p_delta * p_cost) / (p_qty + p_delta), 4)
     END;
  $$;

8.2 Tenant et RBAC

  CREATE TABLE companies (
     id            uuid PRIMARY KEY DEFAULT gen_random_uuid(),

       name         text NOT NULL CHECK (length(btrim(name)) BETWEEN 1 AND 200),
       slug         citext NOT NULL UNIQUE CHECK (slug ~ '^[a-z0-9-]{3,40}$'),
       currency     char(3) NOT NULL DEFAULT 'XOF' CHECK (currency ~ '^[A-Z]{3}$'),
       timezone     text NOT NULL DEFAULT 'UTC',
       settings     jsonb NOT NULL DEFAULT '{}'::jsonb,
       created_at   timestamptz NOT NULL DEFAULT now(),
       updated_at   timestamptz NOT NULL DEFAULT now(),
       deleted_at   timestamptz
  );

  CREATE TABLE users (
       id             uuid PRIMARY KEY DEFAULT gen_random_uuid(),
       company_id     uuid NOT NULL REFERENCES companies(id) ON DELETE RESTRICT,
       email          citext NOT NULL CHECK (email ~ '^[^@[:space:]]+@[^@[:space:]]+[.]
  [^@[:space:]]+$'),
       password_hash text NOT NULL,
       full_name      text NOT NULL CHECK (length(btrim(full_name)) > 0),
       is_active      boolean NOT NULL DEFAULT true,
       last_login_at timestamptz,
       created_at     timestamptz NOT NULL DEFAULT now(),
       updated_at     timestamptz NOT NULL DEFAULT now(),
       deleted_at     timestamptz,
       CONSTRAINT users_tenant_uq UNIQUE (company_id, id),
       CONSTRAINT users_email_uq     UNIQUE (company_id, email)   -- unicite PAR tenant
  );

  CREATE TABLE roles (
       id           uuid PRIMARY KEY DEFAULT gen_random_uuid(),
       company_id   uuid NOT NULL REFERENCES companies(id) ON DELETE RESTRICT,
       name         text NOT NULL CHECK (length(btrim(name)) > 0),
       description text,
       is_system    boolean NOT NULL DEFAULT false,
       created_at   timestamptz NOT NULL DEFAULT now(),
       updated_at   timestamptz NOT NULL DEFAULT now(),
       CONSTRAINT roles_tenant_uq UNIQUE (company_id, id),
       CONSTRAINT roles_name_uq      UNIQUE (company_id, name)
  );

  -- Catalogue GLOBAL : vocabulaire des droits commun a tous les tenants
  CREATE TABLE permissions (
       id         uuid PRIMARY KEY DEFAULT gen_random_uuid(),
       code       text NOT NULL UNIQUE,
       resource text NOT NULL,
       action     text NOT NULL CHECK (action IN ('read','create','update','delete',
                                                   'confirm','cancel','export'))
  );

  CREATE TABLE user_roles (
       user_id      uuid NOT NULL,
       role_id      uuid NOT NULL,
       company_id uuid NOT NULL REFERENCES companies(id) ON DELETE RESTRICT,
       granted_at timestamptz NOT NULL DEFAULT now(),
       PRIMARY KEY (user_id, role_id),
       FOREIGN KEY (company_id, user_id) REFERENCES users (company_id, id)
        ON DELETE CASCADE,
       FOREIGN KEY (company_id, role_id) REFERENCES roles (company_id, id)
        ON DELETE RESTRICT
  );

  CREATE TABLE role_permissions (
       role_id        uuid NOT NULL,

       permission_id uuid NOT NULL REFERENCES permissions(id) ON DELETE RESTRICT,
       company_id     uuid NOT NULL REFERENCES companies(id) ON DELETE RESTRICT,
       PRIMARY KEY (role_id, permission_id),
       FOREIGN KEY (company_id, role_id) REFERENCES roles (company_id, id)
        ON DELETE CASCADE
  );

8.5 Stock — journal, mouvements inverses, CMP

  CREATE TABLE inventory_movements (               -- APPEND-ONLY : source de verite
       id                       uuid PRIMARY KEY DEFAULT gen_random_uuid(),
       company_id               uuid NOT NULL REFERENCES companies(id) ON DELETE RESTRICT,
       product_id               uuid NOT NULL,
       type                     movement_type NOT NULL,
       qty_delta                numeric(14,3) NOT NULL CHECK (qty_delta <> 0),
       unit_cost                numeric(14,4) NOT NULL CHECK (unit_cost >= 0),

       ref_type               text CHECK (ref_type IN ('sale_item','sale_return_item',
                                                        'manual','import','initial')),
       ref_id                 uuid,
       reversal_of_movement_id uuid,                                        -- V1.1
       supplier_id            uuid,
       note                   text,
       occurred_at            timestamptz NOT NULL DEFAULT now(),
       created_by             uuid,
       created_at             timestamptz NOT NULL DEFAULT now(),
       CONSTRAINT inv_tenant_uq UNIQUE (company_id, id),
       CONSTRAINT inv_product_fk FOREIGN KEY (company_id, product_id)
        REFERENCES products (company_id, id) ON DELETE RESTRICT,
       CONSTRAINT inv_supplier_fk FOREIGN KEY (company_id, supplier_id)
        REFERENCES suppliers (company_id, id) ON DELETE RESTRICT,
       CONSTRAINT inv_creator_fk FOREIGN KEY (company_id, created_by)
        REFERENCES users (company_id, id) ON DELETE RESTRICT,
       -- V1.1 : FK auto-referente TENANT-AWARE vers le mouvement annule
       CONSTRAINT inv_reversal_fk FOREIGN KEY (company_id, reversal_of_movement_id)
        REFERENCES inventory_movements (company_id, id) ON DELETE RESTRICT,
       CONSTRAINT inv_reversal_not_self CHECK (reversal_of_movement_id IS DISTINCT FROM id),
       -- V1.1 : un mouvement ne peut etre annule QU'UNE SEULE FOIS
       CONSTRAINT inv_reversal_uq UNIQUE (company_id, reversal_of_movement_id),
       CONSTRAINT inv_sign_coherent CHECK (
        (type IN ('initial','purchase_in','customer_return_in') AND qty_delta > 0)
   OR (type IN ('sale_out','supplier_return_out','loss')           AND qty_delta < 0)
   OR (type = 'adjustment')),
       CONSTRAINT inv_ref_pair CHECK (num_nonnulls(ref_type, ref_id) <> 1)
  );

  -- V1.1 : idempotence des mouvements DIRECTS uniquement. Sans la clause
  -- partielle, le mouvement inverse serait rejete comme doublon de son origine.
  CREATE UNIQUE INDEX inv_ref_idempotence
       ON inventory_movements (company_id, ref_type, ref_id)
       WHERE ref_id IS NOT NULL AND reversal_of_movement_id IS NULL;

  -- ------------------------------------------- immuabilite du journal
  -- V1.1 : exception explicite au lieu d'une RULE "DO INSTEAD NOTHING", qui
  -- masquait silencieusement un bug applicatif en renvoyant "0 ligne modifiee".
  CREATE OR REPLACE FUNCTION inv_immutable() RETURNS trigger AS $$
  BEGIN
       RAISE EXCEPTION 'inventory_movements est append-only : % interdit. '
        'Corriger par un mouvement inverse ou un ajustement.', TG_OP
        USING ERRCODE = 'restrict_violation';
  END $$ LANGUAGE plpgsql;
  CREATE TRIGGER inv_no_update BEFORE UPDATE ON inventory_movements
       FOR EACH ROW EXECUTE FUNCTION inv_immutable();
  CREATE TRIGGER inv_no_delete BEFORE DELETE ON inventory_movements
       FOR EACH ROW EXECUTE FUNCTION inv_immutable();

  -- ------------------------------------------- validation du mouvement inverse
  -- Nommage t10_/t20_ : les triggers BEFORE se declenchent dans l'ordre
  -- alphabetique, la validation doit preceder l'application du CMP.
  CREATE OR REPLACE FUNCTION inv_check_reversal() RETURNS trigger AS $$
  DECLARE o inventory_movements;
  BEGIN
       IF NEW.reversal_of_movement_id IS NULL THEN RETURN NEW; END IF;
       SELECT * INTO o FROM inventory_movements
       WHERE company_id = NEW.company_id AND id = NEW.reversal_of_movement_id
       FOR SHARE;
       IF NOT FOUND THEN
        RAISE EXCEPTION 'Mouvement a annuler introuvable dans ce tenant';

    END IF;
    IF o.reversal_of_movement_id IS NOT NULL THEN
       RAISE EXCEPTION 'On ne peut pas annuler une annulation (mouvement %)', o.id;
    END IF;
    IF o.product_id <> NEW.product_id THEN
       RAISE EXCEPTION 'Le mouvement inverse doit porter sur le meme produit';
    END IF;
    IF NEW.qty_delta <> -o.qty_delta THEN
       RAISE EXCEPTION 'Quantite inverse attendue % , recue %',
          -o.qty_delta, NEW.qty_delta;
    END IF;
    NEW.unit_cost := o.unit_cost;      -- on rend la marchandise a son cout de sortie
    NEW.ref_type   := o.ref_type;      -- tracabilite conservee vers l'objet metier
    NEW.ref_id     := o.ref_id;
    RETURN NEW;
  END $$ LANGUAGE plpgsql;
  CREATE TRIGGER t10_inv_check_reversal BEFORE INSERT ON inventory_movements
    FOR EACH ROW EXECUTE FUNCTION inv_check_reversal();

  -- ------------------------------------------- application : stock + CMP
  CREATE OR REPLACE FUNCTION inv_apply_movement() RETURNS trigger AS $$
  DECLARE
    v_tracked boolean;
    v_qty numeric(14,3); v_avg numeric(14,4);
    v_new_qty numeric(14,3); v_new_avg numeric(14,4);
  BEGIN
    SELECT is_stock_tracked INTO v_tracked FROM products
     WHERE company_id = NEW.company_id AND id = NEW.product_id;
    IF NOT FOUND THEN
       RAISE EXCEPTION 'Produit % inconnu dans ce tenant', NEW.product_id;
    END IF;
    IF NOT v_tracked THEN
       RAISE EXCEPTION 'Produit % non suivi en stock : aucun mouvement autorise',
          NEW.product_id USING ERRCODE = 'restrict_violation';
    END IF;

    INSERT INTO product_stock (product_id, company_id, qty_on_hand, avg_cost)
    VALUES (NEW.product_id, NEW.company_id, 0, 0)
    ON CONFLICT (product_id) DO NOTHING;

    -- verrou de ligne : serialise deux caisses sur le meme article
    SELECT qty_on_hand, avg_cost INTO v_qty, v_avg FROM product_stock
     WHERE product_id = NEW.product_id FOR UPDATE;

    IF NEW.qty_delta < 0 AND NEW.reversal_of_movement_id IS NULL THEN
       -- SORTIE : le cout de sortie est le CMP applicable a cet instant
       NEW.unit_cost := v_avg;
    END IF;

    v_new_qty := v_qty + NEW.qty_delta;
    IF v_new_qty < 0 THEN
       RAISE EXCEPTION
          'Stock negatif interdit : produit %, disponible %, sortie demandee %',
          NEW.product_id, v_qty, abs(NEW.qty_delta)
          USING ERRCODE = 'check_violation';
    END IF;
    IF NEW.qty_delta > 0 AND NEW.unit_cost IS NULL THEN
       RAISE EXCEPTION 'Entree de stock sans cout unitaire : CMP indefinissable';
    END IF;

    v_new_avg := cmp_next_avg(v_qty, v_avg, NEW.qty_delta, NEW.unit_cost);

    UPDATE product_stock
          SET qty_on_hand      = v_new_qty,
             avg_cost         = v_new_avg,
             last_movement_at = GREATEST(COALESCE(last_movement_at, NEW.occurred_at),
                                           NEW.occurred_at)
     WHERE product_id = NEW.product_id;
    RETURN NEW;
  END $$ LANGUAGE plpgsql;
  CREATE TRIGGER t20_inv_apply BEFORE INSERT ON inventory_movements
    FOR EACH ROW EXECUTE FUNCTION inv_apply_movement();

  -- ------------------------------------------- reconstruction (job nocturne)
  -- Rejoue le journal dans l'ordre canonique et applique la MEME formule CMP.
  CREATE OR REPLACE FUNCTION rebuild_product_stock(p_company uuid)
  RETURNS TABLE (product_id uuid, qty_ecart numeric, avg_ecart numeric)
  LANGUAGE plpgsql AS $$
  DECLARE m record; v_qty numeric(14,3); v_avg numeric(14,4); v_prod uuid;
  BEGIN
    CREATE TEMP TABLE _rebuilt (product_id uuid PRIMARY KEY,
       qty numeric(14,3), avg numeric(14,4)) ON COMMIT DROP;
    v_prod := NULL;
    FOR m IN
       SELECT * FROM inventory_movements
          WHERE company_id = p_company
          ORDER BY product_id, occurred_at, id         -- ordre canonique
    LOOP
       IF v_prod IS DISTINCT FROM m.product_id THEN
           IF v_prod IS NOT NULL THEN
            INSERT INTO _rebuilt VALUES (v_prod, v_qty, v_avg);
           END IF;
           v_prod := m.product_id; v_qty := 0; v_avg := 0;
       END IF;
       v_avg := cmp_next_avg(v_qty, v_avg, m.qty_delta, m.unit_cost);
       v_qty := v_qty + m.qty_delta;
       IF v_qty < 0 THEN
           RAISE WARNING 'Etat intermediaire negatif : produit %, mouvement % '
            '(antidatage probable)', m.product_id, m.id;
       END IF;
    END LOOP;
    IF v_prod IS NOT NULL THEN INSERT INTO _rebuilt VALUES (v_prod, v_qty, v_avg); END IF;

    RETURN QUERY
       SELECT r.product_id, r.qty - s.qty_on_hand, r.avg - s.avg_cost
           FROM _rebuilt r JOIN product_stock s USING (product_id)
          WHERE r.qty <> s.qty_on_hand OR r.avg <> s.avg_cost;

    UPDATE product_stock s
          SET qty_on_hand = r.qty, avg_cost = r.avg, recomputed_at = now()
       FROM _rebuilt r
     WHERE s.product_id = r.product_id AND s.company_id = p_company;
  END $$;

  -- ---------------------------------- totaux de vente (subtotal / remise / taxe)
  CREATE OR REPLACE FUNCTION sale_recompute_totals() RETURNS trigger AS $$
  DECLARE v_sale uuid; v_co uuid;
  BEGIN
    v_sale := COALESCE(NEW.sale_id, OLD.sale_id);
    v_co     := COALESCE(NEW.company_id, OLD.company_id);
    UPDATE sales s SET
       subtotal        = COALESCE(t.sub, 0),
       discount_total = COALESCE(t.dis, 0),
       tax_total       = COALESCE(t.tax, 0),
       updated_at      = now()
    FROM (SELECT SUM(qty * unit_price) sub, SUM(discount) dis, SUM(tax_amount) tax
               FROM sale_items
             WHERE company_id = v_co AND sale_id = v_sale) t
    WHERE s.company_id = v_co AND s.id = v_sale;
    RETURN NULL;
  END $$ LANGUAGE plpgsql;
  CREATE TRIGGER sale_items_totals AFTER INSERT OR UPDATE OR DELETE ON sale_items
    FOR EACH ROW EXECUTE FUNCTION sale_recompute_totals();

  -- ---------------------------------- immuabilite des lignes apres confirmation
  CREATE OR REPLACE FUNCTION sale_items_frozen() RETURNS trigger AS $$
  DECLARE v_status sale_status;
  BEGIN
    SELECT status INTO v_status FROM sales
     WHERE company_id = COALESCE(NEW.company_id, OLD.company_id)
          AND id = COALESCE(NEW.sale_id, OLD.sale_id);
    IF v_status <> 'draft' THEN
       RAISE EXCEPTION 'Vente % : les lignes sont figees apres confirmation',
           COALESCE(NEW.sale_id, OLD.sale_id) USING ERRCODE = 'restrict_violation';
    END IF;
    RETURN COALESCE(NEW, OLD);
  END $$ LANGUAGE plpgsql;
  CREATE TRIGGER sale_items_no_change BEFORE UPDATE OR DELETE ON sale_items
    FOR EACH ROW EXECUTE FUNCTION sale_items_frozen();

  -- ---------------------------------- retour : quantite bornee par la vente
  CREATE OR REPLACE FUNCTION return_item_check() RETURNS trigger AS $$
  DECLARE v_sold numeric(14,3); v_already numeric(14,3);
  BEGIN
    SELECT qty INTO v_sold FROM sale_items
     WHERE company_id = NEW.company_id AND id = NEW.sale_item_id FOR UPDATE;
    IF NOT FOUND THEN RAISE EXCEPTION 'Ligne de vente introuvable'; END IF;
    SELECT COALESCE(SUM(qty), 0) INTO v_already FROM sale_return_items
     WHERE company_id = NEW.company_id AND sale_item_id = NEW.sale_item_id
          AND id <> NEW.id;
    IF v_already + NEW.qty > v_sold THEN
       RAISE EXCEPTION 'Retour cumule % superieur a la quantite vendue %',
           v_already + NEW.qty, v_sold USING ERRCODE = 'check_violation';
    END IF;
    RETURN NEW;
  END $$ LANGUAGE plpgsql;
  CREATE TRIGGER sri_check BEFORE INSERT OR UPDATE ON sale_return_items
    FOR EACH ROW EXECUTE FUNCTION return_item_check();

  CREATE OR REPLACE FUNCTION return_refund_status_set() RETURNS trigger AS $$
  BEGIN
    NEW.refund_status := CASE
       WHEN NEW.total_refund = 0                       THEN 'none_due'
       WHEN NEW.refunded_total = 0                     THEN 'pending'
       WHEN NEW.refunded_total < NEW.total_refund      THEN 'partial'

  END $$ LANGUAGE plpgsql;

  -- ---------------------------------- remboursement : plafonds et sens financier
  CREATE OR REPLACE FUNCTION refund_check() RETURNS trigger AS $$
  DECLARE v_due numeric(14,2); v_free numeric(14,2); v_dir cash_direction;
  BEGIN
    SELECT total_refund - refunded_total INTO v_due FROM sale_returns
     WHERE company_id = NEW.company_id AND id = NEW.sale_return_id FOR UPDATE;
    IF NOT FOUND THEN RAISE EXCEPTION 'Retour introuvable'; END IF;
    IF NEW.amount > v_due THEN
       RAISE EXCEPTION 'Remboursement % superieur au reste du %', NEW.amount, v_due
           USING ERRCODE = 'check_violation';
    END IF;
    SELECT direction, amount - COALESCE(
               (SELECT SUM(r.amount) FROM sale_return_refunds r
                 WHERE r.payment_id = p.id AND r.id <> NEW.id), 0)
       INTO v_dir, v_free
       FROM payments p
     WHERE p.company_id = NEW.company_id AND p.id = NEW.payment_id FOR UPDATE;
    IF v_dir <> 'outflow' THEN
       RAISE EXCEPTION 'Un remboursement de retour exige direction = outflow';
    END IF;
    IF NEW.amount > v_free THEN
       RAISE EXCEPTION 'Paiement sur-affecte : reste % disponible', v_free;
    END IF;
    RETURN NEW;
  END $$ LANGUAGE plpgsql;
  CREATE TRIGGER srr_check BEFORE INSERT OR UPDATE ON sale_return_refunds
    FOR EACH ROW EXECUTE FUNCTION refund_check();

  CREATE OR REPLACE FUNCTION refund_rollup() RETURNS trigger AS $$
  DECLARE v_ret uuid; v_co uuid;
  BEGIN
    v_ret := COALESCE(NEW.sale_return_id, OLD.sale_return_id);
    v_co    := COALESCE(NEW.company_id, OLD.company_id);
    UPDATE sale_returns r SET refunded_total = COALESCE(t.s, 0), updated_at = now()
       FROM (SELECT SUM(amount) s FROM sale_return_refunds
               WHERE company_id = v_co AND sale_return_id = v_ret) t
     WHERE r.company_id = v_co AND r.id = v_ret;
    RETURN NULL;

  END $$ LANGUAGE plpgsql;
  CREATE TRIGGER srr_rollup AFTER INSERT OR UPDATE OR DELETE ON sale_return_refunds
    FOR EACH ROW EXECUTE FUNCTION refund_rollup();

  -- ---------------------------------- lettrage paiement / dette (V1, conserve)
  CREATE OR REPLACE FUNCTION allocation_check() RETURNS trigger AS $$
  DECLARE v_bal numeric(14,2); v_free numeric(14,2);
  BEGIN
    SELECT balance INTO v_bal FROM debts
     WHERE company_id = NEW.company_id AND id = NEW.debt_id FOR UPDATE;
    IF NEW.amount > v_bal THEN
       RAISE EXCEPTION 'Affectation % superieure au solde restant %',
           NEW.amount, v_bal USING ERRCODE = 'check_violation';
    END IF;
    SELECT p.amount - COALESCE((SELECT SUM(a.amount) FROM payment_allocations a
                                     WHERE a.payment_id = p.id AND a.id <> NEW.id), 0)
       INTO v_free FROM payments p
     WHERE p.company_id = NEW.company_id AND p.id = NEW.payment_id FOR UPDATE;
    IF NEW.amount > v_free THEN
       RAISE EXCEPTION 'Paiement sur-affecte : reste % disponible', v_free;
    END IF;
    RETURN NEW;
  END $$ LANGUAGE plpgsql;
  CREATE TRIGGER pa_check BEFORE INSERT OR UPDATE ON payment_allocations
    FOR EACH ROW EXECUTE FUNCTION allocation_check();

  CREATE OR REPLACE FUNCTION debt_rollup() RETURNS trigger AS $$
  DECLARE v_debt uuid; v_co uuid;
  BEGIN
    v_debt := COALESCE(NEW.debt_id, OLD.debt_id);
    v_co     := COALESCE(NEW.company_id, OLD.company_id);
    UPDATE debts d SET
       amount_paid = COALESCE(t.s, 0),
       status = CASE WHEN COALESCE(t.s, 0) = 0               THEN 'open'
                      WHEN COALESCE(t.s, 0) < d.principal      THEN 'partially_paid'
                      ELSE 'paid' END::debt_status,
       closed_at = CASE WHEN COALESCE(t.s, 0) >= d.principal THEN now() END,
       updated_at = now()
    FROM (SELECT SUM(amount) s FROM payment_allocations
             WHERE company_id = v_co AND debt_id = v_debt) t
    WHERE d.company_id = v_co AND d.id = v_debt AND d.status <> 'written_off';
    RETURN NULL;
  END $$ LANGUAGE plpgsql;
  CREATE TRIGGER pa_rollup AFTER INSERT OR UPDATE OR DELETE ON payment_allocations
    FOR EACH ROW EXECUTE FUNCTION debt_rollup();

  -- ---------------------------------- plafond de credit client
  CREATE OR REPLACE FUNCTION debt_credit_limit() RETURNS trigger AS $$
  DECLARE v_limit numeric(14,2); v_open numeric(14,2);
  BEGIN
    IF NEW.party_type <> 'customer' THEN RETURN NEW; END IF;
    SELECT credit_limit INTO v_limit FROM customers
     WHERE company_id = NEW.company_id AND id = NEW.customer_id;
    IF v_limit IS NULL THEN RETURN NEW; END IF;
    SELECT COALESCE(SUM(balance), 0) INTO v_open FROM debts
     WHERE company_id = NEW.company_id AND customer_id = NEW.customer_id
          AND status IN ('open','partially_paid') AND id <> NEW.id;
    IF v_open + NEW.principal > v_limit THEN
       RAISE EXCEPTION 'Plafond de credit depasse : % + % > %',
           v_open, NEW.principal, v_limit USING ERRCODE = 'check_violation';
    END IF;

    RETURN NEW;
  END $$ LANGUAGE plpgsql;
  CREATE TRIGGER debts_credit_limit BEFORE INSERT ON debts
    FOR EACH ROW EXECUTE FUNCTION debt_credit_limit();

  -- ---------------------------------- paid_total de la vente
  CREATE OR REPLACE FUNCTION sale_paid_rollup() RETURNS trigger AS $$
  DECLARE v_sale uuid; v_co uuid;
  BEGIN
    v_sale := COALESCE(NEW.sale_id, OLD.sale_id);
    IF v_sale IS NULL THEN RETURN NULL; END IF;
    v_co := COALESCE(NEW.company_id, OLD.company_id);
    UPDATE sales s SET paid_total = COALESCE(t.s, 0), updated_at = now()
       FROM (SELECT SUM(amount) s FROM payments
              WHERE company_id = v_co AND sale_id = v_sale AND direction = 'inflow') t
     WHERE s.company_id = v_co AND s.id = v_sale;
    RETURN NULL;
  END $$ LANGUAGE plpgsql;
  CREATE TRIGGER payments_sale_rollup AFTER INSERT OR UPDATE OR DELETE ON payments
    FOR EACH ROW EXECUTE FUNCTION sale_paid_rollup();

        yhat_lower IS NULL OR yhat_upper IS NULL OR yhat_lower <= yhat_upper),
       -- PostgreSQL 15+ : product_id nullable doit participer a l'unicite

        (company_id, forecast_type, product_id, as_of, horizon_date, model_version)

  -- V1.1 : anomalies, observations, recommandations persistees
  CREATE TABLE insights (
       id              uuid PRIMARY KEY DEFAULT gen_random_uuid(),
       company_id      uuid NOT NULL REFERENCES companies(id) ON DELETE RESTRICT,
       insight_type    insight_type NOT NULL,

       severity       insight_severity NOT NULL DEFAULT 'info',
       title          text NOT NULL CHECK (length(btrim(title)) BETWEEN 1 AND 160),
       body           text,
       entity_table   text CHECK (entity_table IN ('products','customers','suppliers',
                       'expense_categories','sales','users')),
       entity_id      uuid,
       evidence       jsonb NOT NULL DEFAULT '{}'::jsonb,
       dedupe_key     text NOT NULL,
       model_name     text,
       model_version text,
       status         insight_status NOT NULL DEFAULT 'new',
       read_at        timestamptz,
       resolved_at    timestamptz,
       generated_at   timestamptz NOT NULL DEFAULT now(),
       valid_until    timestamptz,
       CONSTRAINT insights_tenant_uq UNIQUE (company_id, id),
       -- reference souple assumee : pas de FK polymorphe, l'insight survit a
       -- l'archivage de sa cible ; les deux colonnes vont par paire
       CONSTRAINT insights_entity_pair CHECK (num_nonnulls(entity_table, entity_id) <> 1),
       CONSTRAINT insights_read_coherent CHECK (status <> 'new' OR read_at IS NULL),
       CONSTRAINT insights_resolved_coherent CHECK (
        (status = 'resolved') = (resolved_at IS NOT NULL))
  );
  -- Idempotence : un rejeu met a jour l'insight vivant. Un insight ecarte par
  -- l'utilisateur ne bloque pas une reemission si le probleme reapparait.
  CREATE UNIQUE INDEX insights_dedupe_uq ON insights (company_id, dedupe_key)
       WHERE status <> 'dismissed';

8.11 Row Level Security — couverture complète

  -- Boucle explicite sur TOUTES les tables tenant-scoped, y compris celles
  -- ajoutees en V1.1 et audit_logs, oublie en V1.
  DO $$
  DECLARE t text;
  BEGIN
    FOREACH t IN ARRAY ARRAY[
       'users','roles','user_roles','role_permissions',
       'customers','suppliers','categories','products','product_prices',
       'product_stock','sales','sale_items','sale_returns','sale_return_items',
       'sale_return_refunds','inventory_movements',
       'expense_categories','expenses','debts','payments','payment_allocations',
       'notifications','audit_logs','business_events',
       'daily_sales_facts','ml_feature_snapshots','forecasts','insights'
    ] LOOP
       EXECUTE format('ALTER TABLE %I ENABLE ROW LEVEL SECURITY', t);
       EXECUTE format('ALTER TABLE %I FORCE       ROW LEVEL SECURITY', t);
       EXECUTE format($f$
          CREATE POLICY %1$I_tenant ON %1$I
            USING        (company_id = current_setting('app.company_id', true)::uuid)
            WITH CHECK (company_id = current_setting('app.company_id', true)::uuid)
       $f$, t);
    END LOOP;
  END $$;

  -- companies : politique sur id, pour empecher l'enumeration des tenants
  ALTER TABLE companies ENABLE ROW LEVEL SECURITY;
  ALTER TABLE companies FORCE      ROW LEVEL SECURITY;
  CREATE POLICY companies_self ON companies
    USING           (id = current_setting('app.company_id', true)::uuid)
    WITH CHECK (id = current_setting('app.company_id', true)::uuid);
  -- permissions : catalogue global, aucune donnee d'entreprise, pas de RLS.

  -- Role applicatif : aucun privilege de contournement, aucun DELETE physique
  CREATE ROLE coges_app LOGIN NOSUPERUSER NOBYPASSRLS NOCREATEDB NOCREATEROLE;
  GRANT USAGE ON SCHEMA public TO coges_app;
  GRANT SELECT, INSERT, UPDATE ON ALL TABLES IN SCHEMA public TO coges_app;
  GRANT SELECT ON permissions TO coges_app;
  GRANT USAGE ON ALL SEQUENCES IN SCHEMA public TO coges_app;

  -- Role analytique : ecrit les projections, lit le transactionnel, pose le
  -- meme SET LOCAL par entreprise. Pas de BYPASSRLS non plus.
  CREATE ROLE coges_analytics LOGIN NOSUPERUSER NOBYPASSRLS;
  GRANT SELECT ON ALL TABLES IN SCHEMA public TO coges_analytics;
  GRANT INSERT, UPDATE, DELETE ON daily_sales_facts, ml_feature_snapshots,
    forecasts, insights TO coges_analytics;

  COMMIT;
